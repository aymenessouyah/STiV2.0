1. Comparaison du stockage : CHAR(10) vs VARCHAR(10)

CREATE TABLE client (
  nom_char    CHAR(10),
  nom_varchar VARCHAR(10)
);
(essayez plus de 10 caractères)

CHAR(10) Longueur fixe, complétée par des espaces (en jaune/orange)

VARCHAR(10) Longueur variable, 1 octet de longueur (en vert) + les caractères utiles

2. Contrainte de domaine : CHECK (note BETWEEN 0 AND 20)

CREATE TABLE etudiant(
  id   INT PRIMARY KEY,
  nom  VARCHAR(10),
  note DECIMAL(4,2),
  CONSTRAINT chk_note CHECK (note BETWEEN 0 AND 20)
);
