Activité 1 : Rôle des méthodes de l'objet Date

  • getDay() : Jour de la semaine (0 pour Dimanche, 1 pour Lundi, ..., 6 pour Samedi).
  • getMonth() : Mois de l'année (0 pour Janvier à 11 pour Décembre).
  • getFullYear() : Année courante sur 4 chiffres (ex: 2026).
  • getHours() / getMinutes() / getSeconds() : Heure, minutes et secondes courantes.
  • getTime() / Date.now() : Temps écoulé en millisecondes depuis le 1er janvier 1970.
  • getDate() : Jour du mois (1 à 31).

Fonctions Utilitaires de Contrôle (Saisie sans Regex)

// 1. Contrôle : Chiffres uniquement
function estChiffres(ch) {
  if (ch.length === 0) return false;
  for (let i = 0; i < ch.length; i++) {
    let c = ch[i];
    if (c < '0' || c > '9') return false;
  }
  return true;
}

// 2. Contrôle : Chaine alphabétique (Lettres + espaces)
function estAlphabetique(ch) {
  if (ch.length === 0) return false;
  let temp = ch.toUpperCase();
  for (let i = 0; i < temp.length; i++) {
    let c = temp[i];
    if ((c < 'A' || c > 'Z') && c !== ' ') return false;
  }
  return true;
}

// 3. Contrôle : Caractère unique est une lettre
function estLettre(c) {
  if (!c || c.length !== 1) return false;
  let temp = c.toUpperCase();
  return (temp >= 'A' && temp <= 'Z');
}

Activité 2 : Calcul d'Âge et Prochain Anniversaire

function calculerAge() {
  let dateSaisie = document.getElementById("datenaissance").value;
  if (!dateSaisie) return alert("Veuillez sélectionner une date !");

  let aujourdhui = new Date();
  let naiss = new Date(dateSaisie);

  let age = aujourdhui.getFullYear() - naiss.getFullYear();
  let m = aujourdhui.getMonth() - naiss.getMonth();
  if (m < 0 || (m === 0 && aujourdhui.getDate() < naiss.getDate())) {
    age--;
  }

  let prochainAnniv = new Date(aujourdhui.getFullYear(), naiss.getMonth(), naiss.getDate());
  if (aujourdhui > prochainAnniv) {
    prochainAnniv.setFullYear(aujourdhui.getFullYear() + 1);
  }

  let diffMs = prochainAnniv.getTime() - aujourdhui.getTime();
  let joursRestants = Math.ceil(diffMs / (1000 * 60 * 60 * 24));

  document.getElementById("resAge").innerText = age + " ans";
  document.getElementById("resJours").innerText = joursRestants + " jours";
}

Activité 4 : Jeu de Devinette (Tableaux & Math)

let listeNombres = [];

function genererListe() {
  let n = parseInt(document.getElementById("nbElements").value);
  if (isNaN(n) || n < 6 || n > 10) return alert("Saisir un nombre entre 6 et 10");

  listeNombres = [];
  while (listeNombres.length < n) {
    let num = Math.floor(Math.random() * 100) + 1;
    if (listeNombres.indexOf(num) === -1) {
      listeNombres.push(num);
    }
  }

  document.getElementById("minVal").value = Math.min.apply(null, listeNombres);
  document.getElementById("maxVal").value = Math.max.apply(null, listeNombres);
}

function deviner() {
  let choix = parseInt(document.getElementById("valeurSaisie").value);
  if (listeNombres.indexOf(choix) !== -1) {
    alert("Vous avez gagné ! Liste : " + listeNombres.join(", "));
  } else {
    alert("Vous avez perdu ! Liste : " + listeNombres.join(", "));
  }
}

Activité 6 : Validation E-mail et Mot de Passe (Sans Regex)

function genererMDP() {
  let email = document.getElementById("email").value;
  let dateN = document.getElementById("dateN").value;

  let posArobase = email.indexOf('@');
  let lastArobase = email.lastIndexOf('@');

  if (posArobase === -1 || posArobase !== lastArobase) {
    return alert("L'adresse e-mail doit contenir un seul symbole @ !");
  }

  let identifiant = email.substring(0, posArobase);

  if (identifiant.length < 6) {
    return alert("L'identifiant doit comporter au moins 6 caractères !");
  }

  let premierCar = identifiant[0];
  let dernierCar = identifiant[identifiant.length - 1];
  if (!estLettre(premierCar) || !estLettre(dernierCar)) {
    return alert("L'identifiant doit commencer et se terminer par une lettre !");
  }

  let nbPoints = 0;
  for (let i = 0; i < identifiant.length; i++) {
    let c = identifiant[i];
    if (c === '.') {
      nbPoints++;
    } else if (!estLettre(c) && (c < '0' || c > '9')) {
      return alert("L'identifiant doit comporter uniquement des caractères alphanumériques et un point !");
    }
  }

  if (nbPoints !== 1) {
    return alert("L'identifiant doit comporter exactement un point !");
  }

  let ch1 = dateN.substring(0, 4);
  let ch2 = identifiant.toUpperCase();
  let ch3 = ch2.substring(0, ch2.length - 1) + 'F';
  let posPoint = email.lastIndexOf('.');

  let mdp = ch1 + ch2 + ch3 + posPoint;
  document.getElementById("resMDP").value = mdp;
}

Activité 7 : Validation Examen (CIN & Noms sans Regex)

function verif() {
  let cin = document.getElementById("cin").value;
  let nom = document.getElementById("nom").value;
  let prenom = document.getElementById("prenom").value;

  if (cin.length !== 8 || !estChiffres(cin)) {
    return alert("Le CIN doit être composé de exactement 8 chiffres !");
  }

  if (!estAlphabetique(nom) || !estAlphabetique(prenom)) {
    return alert("Le Nom et le Prénom doivent être uniquement alphabétiques !");
  }

  let bonnes = 0, mauvaises = 0;

  let q1 = document.querySelector('input[name="q1"]:checked');
  if (q1 && q1.value === "+,-,*,/") bonnes++; else mauvaises++;

  let q2 = document.querySelector('input[name="q2"]:checked');
  if (q2 && q2.value === "MOD") bonnes++; else mauvaises++;

  let q3 = document.getElementById("q3_select").value;
  if (q3 === "2-1-3") bonnes++; else mauvaises++;

  alert("Vous avez " + bonnes + " bonne(s) réponse(s) et " + mauvaises + " mauvaise(s) réponse(s)");
}
