2. Vérification d'existence et insertion d'un élève (inscription.php)

<?php
$con = mysqli_connect("localhost", "root", "", "bd_ecole");

// 1) Récupération des valeurs envoyées par le formulaire HTML (méthode POST)
$id  = $_POST["id"];
$np  = $_POST["np"];
$moy = $_POST["moy"];
$dn  = $_POST["dn"];

// 2) Vérifier si l'identifiant existe déjà dans la table eleves
$reqVerif = "SELECT * FROM eleves WHERE Id = '$id'";
$resVerif = mysqli_query($con, $reqVerif);

if (mysqli_num_rows($resVerif) > 0) {
    echo "Erreur : un élève portant l'identifiant $id est déjà inscrit !";
} else {
    // 3) Insertion du nouvel élève (apostrophes autour des chaînes et dates)
    $reqIns = "INSERT INTO eleves VALUES ('$id', '$np', $moy, '$dn')";
    mysqli_query($con, $reqIns);

    // 4) Contrôle du nombre de lignes insérées avec mysqli_affected_rows($con)
    if (mysqli_affected_rows($con) > 0) {
        $cat = substr($id, 0, 1); // Initiale de l'Id = Catégorie
        echo "Élève inscrit avec succès dans la Catégorie $cat !";
    } else {
        echo "Échec de l'insertion.";
    }
}
mysqli_close($con);
?>

3. Sélection, tri par Catégorie et séparation Nom / Prénom (liste.php)

Rappel de cours (Différence entre les 3 fonctions fetch) :
• mysqli_fetch_row($res) : renvoie un tableau indicé numériquement ($t[0], $t[1], $t[2]).
• mysqli_fetch_assoc($res) : renvoie un tableau associatif indexé par les noms des colonnes ($t["Id"], $t["Nom_prenom"]).
• mysqli_fetch_array($res) : renvoie un tableau acceptant à la fois les indices numériques ET les noms de colonnes. 
<?php
$con = mysqli_connect("localhost", "root", "", "bd_ecole");

// Sélection des élèves admis, triés par Catégorie (1re lettre de Id) puis par Moyenne décroissante
$req = "SELECT Id, Nom_prenom, Moyenne, DateNaissance FROM eleves
        WHERE Moyenne >= 10
        ORDER BY LEFT(Id, 1) ASC, Moyenne DESC";
$res = mysqli_query($con, $req);

if (mysqli_num_rows($res) == 0) {
    echo "Aucun élève admis trouvé.";
} else {
    echo "<table border='1' cellpadding='6'>";
    echo "<tr><th>Catégorie</th><th>Id</th><th>Nom</th><th>Prénom</th><th>Moyenne</th></tr>";

    while ($t = mysqli_fetch_assoc($res)) {
        $cat = substr($t["Id"], 0, 1);           // Initiale de l'Id = Catégorie
        $pos = strpos($t["Nom_prenom"], " ");    // Position du 1er espace
        $nom = substr($t["Nom_prenom"], 0, $pos);
        $prenom = substr($t["Nom_prenom"], $pos + 1);

        echo "<tr>";
        echo "<td>$cat</td>";
        echo "<td>" . $t["Id"] . "</td>";
        echo "<td>$nom</td>";
        echo "<td>$prenom</td>";
        echo "<td>" . $t["Moyenne"] . "</td>";
        echo "</tr>";
    }
    echo "</table>";
}
mysqli_close($con);
?>

4. Comparaison : mysqli_num_rows($res) vs SELECT COUNT(*) AS nb

<?php
$con = mysqli_connect("localhost", "root", "", "bd_ecole");

// --- MÉTHODE A : Avec mysqli_num_rows($res) sur un SELECT classique ---
$r1 = mysqli_query($con, "SELECT Id FROM eleves WHERE Moyenne >= 10");
$nb1 = mysqli_num_rows($r1);
echo "Nombre d'admis (Méthode A) : " . $nb1 . "<br>";

// --- MÉTHODE B : Avec SELECT COUNT(*) AS nb (renvoie toujours 1 seule ligne !) ---
$r2 = mysqli_query($con, "SELECT COUNT(*) AS nb FROM eleves WHERE Moyenne >= 10");
$lig = mysqli_fetch_assoc($r2); // ou mysqli_fetch_row($r2) avec $lig[0]
$nb2 = $lig["nb"];
echo "Nombre d'admis (Méthode B) : " . $nb2 . "<br>";

mysqli_close($con);
?>

5. Mise à jour (UPDATE) et fermeture de la connexion

<?php
$con = mysqli_connect("localhost", "root", "", "bd_ecole");

$reqMaj = "UPDATE eleves SET Moyenne = Moyenne + 0.5 WHERE Id = 'A101'";
mysqli_query($con, $reqMaj);

if (mysqli_affected_rows($con) > 0) {
    echo "Moyenne mise à jour (" . mysqli_affected_rows($con) . " ligne modifiée).";
} else {
    echo "Aucune ligne modifiée (identifiant introuvable).";
}

// Fermeture propre de la connexion MySQL
mysqli_close($con);
?>
