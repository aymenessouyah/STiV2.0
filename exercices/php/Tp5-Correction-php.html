Rappel CSS sur transition :
• all : applique la transition sur toutes les propriétés modifiées au survol.
• ease : démarrage lent, rapide au milieu, fin lente.
• ease-in : démarrage lent puis accélère.
• ease-out : démarrage rapide puis ralentit à la fin.
• ease-in-out : démarrage et fin progressifs et symétriques.

        <p><input type="submit" value="Inscrire"> <input type="reset" value="Annuler"></p>

// Fichier : controle.js (avec let, getElementById uniquement, aucun querySelector)
function estLettreOuEspace(ch) {
    for (let i = 0; i < ch.length; i++) {
        let c = ch[i].toUpperCase();
        if (!((c >= "A" && c <= "Z") || c === " ")) {
            return false;
        }
    }
    return true;
}

function nettoyerEspaces(ch) {
    let s = ch.trim();
    while (s.indexOf("  ") !== -1) {
        s = s.replace("  ", " ");
    }
    return s;
}

function verif() {
    let id  = document.getElementById("id").value.trim();
    let np  = nettoyerEspaces(document.getElementById("np").value);
    document.getElementById("np").value = np; // Réinjecte la chaîne sans espaces superflus
    let moy = document.getElementById("moy").value.trim();
    let dn  = document.getElementById("dn").value;
    let msg = document.getElementById("msg");
    msg.innerHTML = "";

    // 1) Contrôle de Id : commence par une lettre majuscule (= Catégorie)
    if (id.length < 2 || id[0] < "A" || id[0] > "Z") {
        msg.innerHTML = "Erreur : Id doit commencer par une lettre majuscule (Catégorie) !";
        alert("Erreur : Id invalide !");
        return false;
    }

    // 2) Contrôle de Nom_prenom : lettres et espaces uniquement, entre 2 et 5 mots
    let mots = np === "" ? [] : np.split(" ");
    if (!estLettreOuEspace(np) || mots.length < 2 || mots.length > 5) {
        msg.innerHTML = "Erreur : Nom_prenom ne doit contenir que des lettres/espaces (entre 2 et 5 mots) !";
        return false;
    }

    // 3) Contrôle de Moyenne : nombre réel entre 0 et 20
    if (moy === "" || isNaN(moy) || Number(moy) < 0 || Number(moy) > 20) {
        msg.innerHTML = "Erreur : la moyenne doit être comprise entre 0 et 20 !";
        return false;
    }

    // 4) Contrôle de DateNaissance : getMonth() renvoie 0..11 donc + 1
    if (dn === "") {
        msg.innerHTML = "Veuillez saisir la date de naissance !";
        return false;
    }
    let d = new Date(dn);
    let annee = d.getFullYear();
    let mois  = d.getMonth() + 1;
    if (annee < 2004 || annee > 2010) {
        msg.innerHTML = "Erreur : année de naissance hors intervalle (2004-2010), mois lu = " + mois;
        return false;
    }

    return true;
}

<?php
// === 1. Fichier : inscription.php (3 pts) ===
$con = mysqli_connect("localhost", "root", "", "bd_lycee"); // Connexion au serveur MySQL

$id  = $_POST["id"];  // Lecture de l'identifiant
$np  = $_POST["np"];  // Lecture de Nom_prenom
$moy = $_POST["moy"]; // Lecture de la moyenne
$dn  = $_POST["dn"];  // Lecture de la date de naissance

// Vérifier si l'élève existe déjà avec mysqli_num_rows($res)
$req1 = "SELECT * FROM eleves WHERE Id = '$id'";
$res1 = mysqli_query($con, $req1);

if (mysqli_num_rows($res1) > 0) {
    echo "Erreur : l'élève d'identifiant $id existe déjà !";
} else {
    $req2 = "INSERT INTO eleves VALUES ('$id', '$np', $moy, '$dn')";
    mysqli_query($con, $req2);

    if (mysqli_affected_rows($con) > 0) {
        $cat = substr($id, 0, 1); // Initiale de l'Id = Catégorie
        echo "Inscription réussie dans la Catégorie $cat !";
    } else {
        echo "Échec de l'insertion.";
    }
}
mysqli_close($con);
?>
<?php
// === 2. Fichier : recherche.php (2 pts) ===
$con = mysqli_connect("localhost", "root", "", "bd_lycee");
$cat = $_POST["cat"]; // Lettre majuscule choisie (ex: 'A')

$req = "SELECT Id, Nom_prenom, Moyenne, DateNaissance FROM eleves
        WHERE LEFT(Id, 1) = '$cat' ORDER BY Moyenne DESC";
$res = mysqli_query($con, $req);

if (mysqli_num_rows($res) == 0) {
    echo "Aucun élève trouvé dans la catégorie $cat.";
} else {
    echo "<h3>Élèves de la Catégorie $cat :</h3>";
    while ($t = mysqli_fetch_assoc($res)) {
        echo $t["Id"] . " — " . $t["Nom_prenom"] . " (" . $t["Moyenne"] . ")<br>";
    }
}
mysqli_close($con);
?>
<?php
// === 3. Fichier : statistiques.php (3 pts) ===
$con = mysqli_connect("localhost", "root", "", "bd_lycee");

// a) Nombre d'admis avec SELECT COUNT(*) AS nb (renvoie 1 seule ligne -> fetch !)
$resNb = mysqli_query($con, "SELECT COUNT(*) AS nb FROM eleves WHERE Moyenne >= 10");
$ligNb = mysqli_fetch_assoc($resNb);
echo "<h3>Nombre d'élèves admis : " . $ligNb["nb"] . "</h3>";

// b) Liste triée par Catégorie (initiale de Id) avec séparation Nom et Prénom par l'espace
$req = "SELECT Id, Nom_prenom, Moyenne, DateNaissance FROM eleves
        ORDER BY LEFT(Id, 1) ASC, Nom_prenom ASC";
$res = mysqli_query($con, $req);

echo "<table border='1' cellpadding='6'>";
echo "<tr><th>Catégorie</th><th>Id</th><th>Nom</th><th>Prénom</th><th>Moyenne</th></tr>";

while ($t = mysqli_fetch_assoc($res)) {
    $cat    = substr($t["Id"], 0, 1);        // 1re lettre de Id = Catégorie
    $pos    = strpos($t["Nom_prenom"], " "); // Position du 1er espace
    $nom    = substr($t["Nom_prenom"], 0, $pos);
    $prenom = substr($t["Nom_prenom"], $pos + 1);

    echo "<tr>";
    echo "<td>$cat</td><td>" . $t["Id"] . "</td>";
    echo "<td>$nom</td><td>$prenom</td>";
    echo "<td>" . $t["Moyenne"] . "</td>";
    echo "</tr>";
}
echo "</table>";
mysqli_close($con);
?>
